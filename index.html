const formats = [ ZXing.BarcodeFormat.QR_CODE, ZXing.BarcodeFormat.DATA_MATRIX, ZXing.BarcodeFormat.EAN_13, ZXing.BarcodeFormat.EAN_8, ZXing.BarcodeFormat.CODE_128, ZXing.BarcodeFormat.CODE_39, ZXing.BarcodeFormat.UPC_A, ZXing.BarcodeFormat.UPC_E, ZXing.BarcodeFormat.ITF, ZXing.BarcodeFormat.PDF_417, ZXing.BarcodeFormat.AZTEC, ZXing.BarcodeFormat.CODABAR ]; return result.getText(); } catch (e) { return null; } } // Специальная обработка для DataMatrix (адаптивная бинаризация) function preprocessForDataMatrix(imgElement) { const canvas = document.createElement('canvas'); const ctx = canvas.getContext('2d'); canvas.width = imgElement.width; canvas.height = imgElement.height; ctx.drawImage(imgElement, 0, 0); const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height); const data = imageData.data; // Адаптивная пороговая обработка (разбиваем на блоки) const blockSize = 32; const width = canvas.width; const height = canvas.height; // Создаём массив яркости const luminance = []; for (let i = 0; i < data.length; i += 4) { const r = data[i]; const g = data[i + 1]; const b = data[i + 2]; luminance.push((r + g + b) / 3); } // Для каждого пикселя вычисляем локальный порог for (let y = 0; y < height; y++) { for (let x = 0; x < width; x++) { const idx = y * width + x; // Суммируем яркость в блоке let sum = 0; let count = 0; const x1 = Math.max(0, x - blockSize / 2); const x2 = Math.min(width, x + blockSize / 2); const y1 = Math.max(0, y - blockSize / 2); const y2 = Math.min(height, y + blockSize / 2); for (let by = y1; by < y2; by++) { for (let bx = x1; bx < x2; bx++) { sum += luminance[by * width + bx]; count++; } } const localThreshold = sum / count - 15; // небольшой сдвиг const bw = luminance[idx] > localThreshold ? 255 : 0; const pixelIdx = idx * 4; data[pixelIdx] = bw; data[pixelIdx + 1] = bw; data[pixelIdx + 2] = bw; } } ctx.putImageData(imageData, 0, 0); return canvas; } // Усиление контраста с сохранением структуры (для QR и штрих-кодов) function enhanceContrast(imgElement) { const canvas = document.createElement('canvas'); const data = imageData.data; // Находим min и max яркости для полного растяжения гистограммы let minBrightness = 255; let maxBrightness = 0; for (let i = 0; i < data.length; i += 4) { const brightness = (data[i] + data[i + 1] + data[i + 2]) / 3; minBrightness = Math.min(minBrightness, brightness); maxBrightness = Math.max(maxBrightness, brightness); } // Растягиваем контраст на весь диапазон const range = maxBrightness - minBrightness; if (range > 0) { for (let i = 0; i < data.length; i += 4) { const brightness = (data[i] + data[i + 1] + data[i + 2]) / 3; const newBrightness = ((brightness - minBrightness) / range) * 255; data[i] = newBrightness; data[i + 1] = newBrightness; data[i + 2] = newBrightness; } } ctx.putImageData(imageData, 0, 0); return canvas; } // Собираем варианты для попыток (в порядке эффективности) const attempts = []; // 1.
